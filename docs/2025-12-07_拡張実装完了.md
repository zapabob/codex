`yaml
servers:
  serena:
    command: npx
    args: [--port, 3001]
    env: {MCP_SERENA_MODEL: gpt-4o}
    capabilities: [code-analysis, semantic-search]

development:
  parallel-dev: [serena, github, git-enhanced]
  centralized-dev: [serena, arxiv, docs-analyzer]
