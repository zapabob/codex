## 🔧 使用方法

### GPU検証実行
```rust
use codex_core::qc::gpu_validation::{GPUValidationSuite, GPUValidationConfig};

// GPU検証設定
let config = GPUValidationConfig {
    enable_cuda_validation: true,
    enable_performance_benchmarking: true,
    enable_memory_validation: true,
    enable_kernel_validation: true,
    benchmark_iterations: 10,
    validation_timeout_seconds: 300,
    min_performance_threshold: 30.0,
    memory_stress_test_mb: 4096,
};

// 検証実行
let suite = GPUValidationSuite::new(config);
let report = suite.run_full_validation().await?;

// 結果確認
println!("CUDA検知: {}", report.cuda_detected);
println!("互換デバイス数: {}", report.compatible_devices.len());
println!("平均性能: {:.1} GFLOPS", report.performance_summary.average_gflops);
```

### インターフェーステスト実行
```rust
use codex_core::qc::gui_testing::{InterfaceTestSuite, InterfaceTestConfig};

// テスト設定
let config = InterfaceTestConfig {
    enable_gui_testing: true,
    enable_tui_testing: true,
    enable_cli_testing: true,
    enable_integration_testing: true,
    test_timeout_seconds: 60,
    max_concurrent_tests: 5,
    screenshot_on_failure: true,
    record_video: false,
    performance_thresholds: /* ... */,
};

// テストスイート作成
let suite = InterfaceTestSuite::new(config);

// macOSテストシナリオ生成
let scenarios = suite.generate_macos_test_scenarios().await;
for scenario in scenarios {
    suite.add_test_case(scenario).await;
}

// 全テスト実行
let results = suite.run_all_tests().await?;
println!("合格: {}/{}", results.passed_tests, results.total_tests);
```
