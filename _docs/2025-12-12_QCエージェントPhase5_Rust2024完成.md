## 🔧 使用方法

### GPU/量子最適化
```rust
use codex_core::qc::gpu_acceleration::{GpuStatisticalAnalyzer, QuantumOptimizer};

// GPU統計分析
let gpu_analyzer = GpuStatisticalAnalyzer::new()?;
let result = gpu_analyzer.gpu_anova_test(&samples).await?;

// 量子最適化
let quantum_optimizer = QuantumOptimizer::new()?;
let qaoa_result = quantum_optimizer.gpu_qaoa_maxcut(&adjacency_matrix).await?;
```

### 多AI並列実行
```rust
use codex_core::qc::multi_ai_orchestration::{MultiAIOrchestrator, ConsensusAlgorithm};

// 多AIコンセンサス実行
let orchestrator = MultiAIOrchestrator::<3, 10>::new(config);
let consensus = orchestrator.execute_with_consensus(task, ConsensusAlgorithm::QualityWeighted, 3).await?;
```

### GUI/CLI連接
```rust
use codex_core::qc::gui_integration::GUICLIBridge;

// GUIコマンド実行
let bridge = GUICLIBridge::new(config);
let result = bridge.execute_command(GUICommandRequest {
    command: "cargo build".to_string(),
    internet_access: true,
    sandbox_mode: SandboxMode::MacOSSandbox,
    ..
}).await?;
```

### AIセキュリティ
```rust
use codex_core::qc::sandboxed_execution::{AISandboxExecutor, DefaultAISecurityAnalyzer};

// AIセキュリティ分析
let analyzer = Arc::new(DefaultAISecurityAnalyzer::new());
let executor = AISandboxExecutor::new(config, analyzer);
let result = executor.execute_sandboxed(request).await?;
```
